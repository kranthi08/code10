SELECT 
    cc.customer_id AS customerId,
    cc.claim_id AS claimId,
    ccp.ingredient_cost_basis_cd AS ingredientCostBasisCd,
    ccp.ingredient_cost_amt AS ingredientCostAmt,
    ccp.ingredient_id AS ingredientId,
    ccp.compound_product_id_type_cd AS compoundProductIdTypeCd,
    ccp.plan_ingredient_cost_amt AS planIngredientCostAmt,
    ccp.ingredient_quantity_cnt AS ingredientQuantityCnt,
    ccp.ingredient_postpay_ind_3 AS ingredientPostpayInd3,
    ccp.ingredient_postpay_ind_4 AS ingredientPostpayInd4,
    ccp.claim_compound_product_sequence_id AS claimCompoundProductSequenceId
FROM 
    claim_core cc
 JOIN 
    claim_compound_product ccp
    ON cc.customer_id = ccp.customer_id
    AND cc.claim_id = ccp.claim_id
LEFT JOIN 
    claim_compound_client_pricing ccpp
    ON ccp.customer_id = ccpp.customer_id
    AND ccp.claim_id = ccpp.claim_id
    AND ccp.claim_compound_product_sequence_id = ccpp.claim_compound_product_sequence_id
LEFT JOIN 
    claim_compound_ndc_details cnd
    ON ccp.customer_id = cnd.customer_id
    AND ccp.claim_id = cnd.claim_id
    AND ccp.claim_compound_product_sequence_id = cnd.claim_compound_product_sequence_id
WHERE 
    cc.customer_id = 319 and cc.claim_id = '266360008701'
ORDER BY 
    ccp.claim_compound_product_sequence_id;
